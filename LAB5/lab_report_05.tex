\section{Name}

\textbf{Morphological Operations, Image Interpolation, and Otsu Segmentation in MATLAB}\\[0.3em]
Topics covered: Morphological Dilation, Erosion, Opening, Closing, Nearest Neighbor Interpolation, Bilinear Interpolation, and Otsu Threshold Segmentation.

%----------------------------------------

\section{Description}

This lab explores mathematical morphology, spatial interpolation techniques, and global image thresholding in MATLAB.

\begin{enumerate}
  \item \textbf{Morphological Dilation:} Expands the boundaries of foreground objects using a 3$\times$3 square structuring element via imdilate, filling small holes and connecting adjacent components.

  \item \textbf{Morphological Erosion:} Shrinks foreground objects by stripping away boundary pixels with imerode, attenuating thin structures and eliminating isolated background noise.

  \item \textbf{Morphological Opening:} Performs erosion followed by dilation using imopen to suppress small bright artifacts without altering the overall shape and size of larger structures.

  \item \textbf{Morphological Closing:} Performs dilation followed by erosion using imclose to bridge narrow gaps, join broken contours, and fill small dark inclusions.

  \item \textbf{Nearest Neighbor Interpolation:} Enlarges image dimensions using imresize by copying the nearest pixel value, which is computationally fast but can produce blocky pixelation.

  \item \textbf{Bilinear Interpolation:} Scales the image by computing a distance-weighted average of the four surrounding pixels, resulting in smoother transitions.

  \item \textbf{Otsu's Threshold Segmentation:} Employs graythresh to automatically find an optimal global threshold that maximizes inter-class variance, segmenting the image into binary foreground and background via im2bw.
\end{enumerate}

\section{Code}

\begin{lstlisting}[style=matlabstyle]
% read image
I = imread('text.png');
BW = im2bw(I);
SE = strel('square', 3);

% 1. dilation
D = imdilate(BW, SE);

% 2. erosion
E = imerode(BW, SE);

% 3. opening
O = imopen(BW, SE);

% 4. closing
C = imclose(BW, SE);

figure;
subplot(2,3,1);
imshow(I);
title('original image');

subplot(2,3,2);
imshow(BW);
title('binary image');

subplot(2,3,3);
imshow(D);
title('dilated image');

subplot(2,3,4);
imshow(E);
title('eroded image');

subplot(2,3,5);
imshow(O);
title('opened image');

subplot(2,3,6);
imshow(C);
title('Closed image');

% read cameraman image
I = imread('cameraman.tif');

% 5. pixel replication / nearest Neighbor
Z_nearest = imresize(I, 2, 'nearest');

% 6. bilinear interpolation
Z_bilinear = imresize(I, 2, 'bilinear');

% 7. otsu threshold segmentation
if size(I,3) == 3
    I = rgb2gray(I);
end
level = graythresh(I);
BW_otsu = im2bw(I, level);

figure;
subplot(2,3,1);
imshow(I);
title('original image');

subplot(2,3,2);
imshow(Z_nearest);
title('nearest neighbor');

subplot(2,3,3);
imshow(Z_bilinear);
title('bilinear interpolation');

subplot(2,3,4);
imshow(BW_otsu);
title('otsu segmentation');
\end{lstlisting}
